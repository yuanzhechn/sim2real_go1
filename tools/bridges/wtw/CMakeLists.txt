cmake_minimum_required(VERSION 3.12)
project(go1_wtw_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -O2")
enable_testing()
add_executable(test_control_gate test_control_gate.cpp)
add_test(NAME control_gate COMMAND test_control_gate)
option(BUILD_LCM_BRIDGE "Build simulator and optional SDK bridge" ON)
set(WTW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../walk-these-ways" CACHE PATH "Sibling upstream WTW clone")
set(UNITREE_SDK_ROOT "" CACHE PATH "Existing Unitree SDK; empty means simulator only")
if(BUILD_LCM_BRIDGE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LCM REQUIRED IMPORTED_TARGET lcm)
  find_program(LCM_GEN lcm-gen REQUIRED)
  set(GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
  file(MAKE_DIRECTORY ${GENERATED})
  set(SCHEMAS)
  foreach(NAME pd_tau_targets_lcmt leg_control_data_lcmt state_estimator_lcmt rc_command_lcmt)
    list(APPEND SCHEMAS ${WTW_ROOT}/go1_gym_deploy/lcm_types/${NAME}.lcm)
  endforeach()
  list(APPEND SCHEMAS ${CMAKE_CURRENT_SOURCE_DIR}/go1_bridge_status_lcmt.lcm)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SCHEMAS})
  execute_process(COMMAND ${LCM_GEN} -x --cpp-hpath ${GENERATED} ${SCHEMAS} RESULT_VARIABLE GEN_RESULT)
  if(NOT GEN_RESULT EQUAL 0)
    message(FATAL_ERROR "LCM generation failed; clone WTW beside sim2real_go1")
  endif()
  execute_process(COMMAND ${LCM_GEN} -p --ppath ${GENERATED} ${CMAKE_CURRENT_SOURCE_DIR}/go1_bridge_status_lcmt.lcm)
  add_executable(go1_wtw_bridge bridge.cpp)
  target_include_directories(go1_wtw_bridge PRIVATE ${GENERATED})
  target_link_libraries(go1_wtw_bridge PRIVATE PkgConfig::LCM pthread)
  if(UNITREE_SDK_ROOT)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
      set(SDK_ARCH arm64)
    else()
      set(SDK_ARCH amd64)
    endif()
    target_compile_definitions(go1_wtw_bridge PRIVATE HAVE_UNITREE_SDK)
    target_include_directories(go1_wtw_bridge PRIVATE ${UNITREE_SDK_ROOT}/include)
    target_link_libraries(go1_wtw_bridge PRIVATE ${UNITREE_SDK_ROOT}/lib/cpp/${SDK_ARCH}/libunitree_legged_sdk.a)
  endif()
endif()
